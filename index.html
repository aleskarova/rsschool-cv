Code Example

Grouping friends by their current online status

const whosOnline = (friends) => {
  const statuses = {};

  for (const friend of friends) {
    const isAway = friend.status === 'online'
      && friend.lastActivity > 10;
    const status = isAway ? 'away' : friend.status;

    if (!statuses[status]) {
      statuses[status] = [];
    }

    statuses[status].push(friend.username);
  }

  return statuses;
};
